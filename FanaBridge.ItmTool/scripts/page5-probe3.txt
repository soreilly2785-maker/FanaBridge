# Page 5 probe, take 3 -- key insight from probe2: slot 0x83 / handle 3
# renders at the RL position (not FR). Probe1 was sending paramId 45 (FR)
# to handle 3, but the firmware may reject mismatched paramIds. Try:
#   handle 3 + paramId 48 (TYRE_RL) -- expect RL to populate
#   handle 4 + paramId 45 (TYRE_FR) -- expect FR to populate
# FL=85 and RR=80 kept as steady reference.
# Run with: script scripts\page5-probe3.txt

connect
activate on

pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
sleep 200

pageset 3 5
paramdefs 82 00 00 - 83 00 00 - 84 00 00 - 85 00 00 -
keepalive
sleep 200
value 0 1 i16 62
value 1 4 u8 4
value 2 42 u8 85
value 5 51 u8 80
sleep 2000

# --- Try RL at handle 3 (paramId 48) and FR at handle 4 (paramId 45) ---
value 3 48 u8 75
value 4 45 u8 90
sleep 4000

# --- Change both to confirm which field is which ---
value 3 48 u8 45
value 4 45 u8 60
sleep 4000
