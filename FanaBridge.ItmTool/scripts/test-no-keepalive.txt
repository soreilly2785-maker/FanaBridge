# No-keepalive endurance test.
# Run with: script scripts\test-no-keepalive.txt
#
# Activates ITM, sets up Page 1, and sends a changing LAP value every 5s
# for ~50s -- WITHOUT ever sending a keepalive. Watch whether the display
# stays live and updates the whole time, or goes blank/frozen/reverts to
# the firmware UI at some point.

connect
activate on
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -

value 0 1 i16 123
value 1 4 u8 3
value 2 505 u8 1
value 3 501 u8 2
value 4 509 f32 65.4
value 5 510 f32 64.1

sleep 5000
value 2 505 u8 2
sleep 5000
value 2 505 u8 3
sleep 5000
value 2 505 u8 4
sleep 5000
value 2 505 u8 5
sleep 5000
value 2 505 u8 6
sleep 5000
value 2 505 u8 7
sleep 5000
value 2 505 u8 8
sleep 5000
value 2 505 u8 9
sleep 5000
value 2 505 u8 10
