# Page 2 (Fuel/ERS/DRS) field-mapping probe.
# Run with: script scripts\page2-probe.txt
#
# Defines all 5 documented Page 2 fields (FUEL, ERS_LEVEL, DRS_ZONE,
# DRS_ACTIVE, DELTA_OWN_BEST) in one ParamDefs call -- GTSWX's detection
# signature for this page requires all 5 present together, so the firmware
# may only show the DRS dots / delta field once all 5 slots are registered.
# Handles continue sequentially from the confirmed FUEL=8/ERS_LEVEL=9.
#
# Watch the screen at each step. Note which step (if any) changes the DRS
# dots on the left, or makes a delta-looking field appear/change.

connect
activate on
pageset 3 2
paramdefs 88 00 00 /50 88 01 00 - 88 02 00 - 88 03 00 - 88 04 00 -

# Baseline: FUEL=25/50, ERS_LEVEL=80
value 0 1 i16 123
value 1 4 u8 3
value 8 5 f32 25.0
value 9 9 i32 80
sleep 3000

# --- Step 1: DRS_ZONE = 1 (handle 10, posLo 2) ---
value 10 14 u8 1
sleep 3000

# --- Step 2: DRS_ZONE = 0 ---
value 10 14 u8 0
sleep 3000

# --- Step 3: DRS_ACTIVE = 1 (handle 11, posLo 3) ---
value 11 15 u8 1
sleep 3000

# --- Step 4: DRS_ACTIVE = 0 ---
value 11 15 u8 0
sleep 3000

# --- Step 5: DELTA_OWN_BEST = -1.234 (handle 12, posLo 4) ---
value 12 516 f32 -1.234
sleep 3000

# --- Step 6: DELTA_OWN_BEST = +0.567 ---
value 12 516 f32 0.567
sleep 4000
