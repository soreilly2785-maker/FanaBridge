# Keepalive validation test.
# Run with: script scripts\test-keepalive.txt
# Watch the display at each step described in the comments.

connect
activate on
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -

value 0 1 i16 123
value 1 4 u8 3
value 2 505 u8 5
value 3 501 u8 2
value 4 509 f32 65.4
value 5 510 f32 64.1

# --- Step 1: confirm the page renders correctly with keepalive running ---
auto on
sleep 3000

# --- Step 2: stop keepalive, then watch the display for several seconds ---
# Does it freeze, go blank, or revert to the firmware's normal UI?
auto off
sleep 5000

# --- Step 3: send a fresh ValueUpdate with keepalive OFF ---
# Does this value actually render on screen, or is it silently ignored?
value 2 505 u8 9
sleep 3000

# --- Step 4: restart keepalive and confirm the display recovers ---
auto on
sleep 3000
auto off
