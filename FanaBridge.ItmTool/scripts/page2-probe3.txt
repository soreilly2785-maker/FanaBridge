# Page 2 field-mapping probe, take 3 -- bootstrap Page 1 first (as our
# production driver does), then switch to Page 2 with the full 5-entry
# layout (FUEL, ERS_LEVEL, DRS_ZONE, DRS_ACTIVE, DELTA_OWN_BEST).
# Run with: script scripts\page2-probe3.txt
#
# Step 1: bootstrap Page 1 (registers the persistent SPEED/GEAR header).
# Step 2: switch to Page 2, 5-entry ParamDefs, send all 5 values.
#         Confirm SPEED/GEAR/FUEL/ERS render -- if they do now (unlike
#         probe2), that confirms the bootstrap was the missing piece.
# Step 3-6: toggle DRS_ZONE / DRS_ACTIVE / DELTA individually.

connect
activate on

# --- Step 1: Page 1 bootstrap ---
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
sleep 1000

# --- Step 2: switch to Page 2, full 5-entry layout ---
pageset 3 2
paramdefs 88 00 00 /50 88 01 00 - 88 02 00 - 88 03 00 - 88 04 00 -
value 0 1 i16 123
value 1 4 u8 3
value 8 5 f32 25.0
value 9 9 i32 80
value 10 14 u8 0
value 11 15 u8 0
value 12 516 f32 0.0
sleep 4000

# --- Step 3: DRS_ZONE = 1 ---
value 10 14 u8 1
sleep 3000

# --- Step 4: DRS_ACTIVE = 1 ---
value 11 15 u8 1
sleep 3000

# --- Step 5: DELTA_OWN_BEST = -1.234 ---
value 12 516 f32 -1.234
sleep 3000

# --- Step 6: DELTA_OWN_BEST = +0.567 ---
value 12 516 f32 0.567
sleep 4000
