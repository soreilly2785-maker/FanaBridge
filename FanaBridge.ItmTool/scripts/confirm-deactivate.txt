# Confirms whether activate off actually works on the PBME.
# Expect: values show, then display reverts to firmware mode (clock/speed etc)
# after activate off. If display just freezes/stays the same, activate off
# is being ignored.
# Run with: script scripts\confirm-deactivate.txt

connect
activate on
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
keepalive
sleep 100
keepalive
sleep 100
keepalive
sleep 100
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
sleep 500

value 0 1 i16 65
value 1 4 u8 3
value 2 505 u8 5
value 3 501 u8 2
value 4 509 f32 65.4
value 5 510 f32 64.1
sleep 3000

# --- Does the display revert to firmware mode here? ---
activate off
sleep 5000

# --- If still ITM, activate on didn't clear; if firmware mode, it worked ---
activate on
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
keepalive
sleep 500
value 0 1 i16 99
value 1 4 u8 9
sleep 3000
