# Page 4 ("Lap Times") verification probe.
# Sends known non-zero values to all four handles to confirm the mapping
# and verify that f32 values render (not blank) when non-zero.
# Run with: script scripts\page4-probe.txt

connect
activate on

# Boot on page 1 first (matches what the production driver does)
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
sleep 200

# Switch to page 4
pageset 3 4
paramdefs 88 00 00 - 88 01 00 - 88 02 00 - 88 03 00 -
keepalive
sleep 500

# Send speed/gear header (handles 0/1) plus all four page 4 fields:
#   h0 = Speed (i16)
#   h1 = Gear  (u8)
#   h2 = LastLapTime  (f32, paramId 510 = 0x01FE) -- expect ~1:35.50
#   h3 = BestLapTime  (f32, paramId 511 = 0x01FF) -- expect ~1:34.20
#   h4 = CarAhead     (f32, paramId 519 = 0x0207) -- expect 2.35s gap
#   h5 = CarBehind    (f32, paramId 520 = 0x0208) -- expect 1.87s gap
value 0 1 i16 65
value 1 4 u8 3
value 2 510 f32 95.5
value 3 511 f32 94.2
value 4 519 f32 2.35
value 5 520 f32 1.87
sleep 4000

# --- Confirm zero renders as blank ---
# Change all four to 0.0 to see what the firmware shows for zero values
value 2 510 f32 0.0
value 3 511 f32 0.0
value 4 519 f32 0.0
value 5 520 f32 0.0
sleep 4000

# --- Restore non-zero values to confirm the display recovers ---
value 2 510 f32 95.5
value 3 511 f32 94.2
value 4 519 f32 2.35
value 5 520 f32 1.87
sleep 4000
