# Page 2 probe, take 6 -- revisit DRS_ZONE / DRS_ACTIVE / DELTA_OWN_BEST now
# that we know Page 2 needs its own keepalive "kick" + short settle + resend
# before its fields render (handles 6/7 for SPEED/GEAR needed exactly this).
# Earlier probes (2/3) never sent a keepalive at all, which may be why
# handles 10/11/12 never appeared to do anything.
# Run with: script scripts\page2-probe6.txt
#
# Step 1: Page 1 bootstrap (as production driver does).
# Step 2: switch to Page 2, full 5-entry ParamDefs (positions 0-4 ->
#         handles 8-12), kick, settle, then resend ALL values including
#         the header (6/7) and FUEL/ERS (8/9) as a baseline.
# Step 3-6: toggle DRS_ZONE / DRS_ACTIVE / DELTA individually.

connect
activate on

# --- Step 1: Page 1 bootstrap ---
pageset 3 1

sleep 200

value 10 14 u8 0

# --- Step 5: DRS_ACTIVE = 0, DELTA_OWN_BEST = -1.234 ---

# --- Step 6: DELTA_OWN_BEST = +0.567 ---
