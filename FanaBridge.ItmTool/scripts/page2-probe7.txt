# Page 2 probe, take 7 -- handles 10/11/12 (positions 2/3/4, the "+8" scheme
# that worked for FUEL/ERS at positions 0/1) showed no effect for DRS_ZONE/
# DRS_ACTIVE/DELTA_OWN_BEST even with the kick+resend pattern that fixed
# SPEED/GEAR. Try two alternative handle schemes for those three fields:
#   A) handles 11/12/13 (off-by-one from the +8 scheme)
#   B) handles 2/3/4 (Page 4's low-handle scheme, ignoring slot 0x88's
#      position-based numbering for these icon-type fields)
# Each block re-kicks + resends FUEL/ERS/SPEED/GEAR alongside the candidate
# handles, since the working fields give us a clear "page is still alive"
# reference point.
# Run with: script scripts\page2-probe7.txt

connect
activate on

# --- Page 1 bootstrap ---
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
sleep 200

# --- Switch to Page 2, full 5-entry layout, kick + settle ---
pageset 3 2
paramdefs 88 00 00 /50 88 01 00 - 88 02 00 - 88 03 00 - 88 04 00 -
keepalive
sleep 200

# --- Block A: handles 11/12/13 for DRS_ZONE/DRS_ACTIVE/DELTA ---
value 6 1 i16 30
value 7 4 u8 6
value 8 5 f32 29.0
value 9 9 i32 60
value 11 14 u8 1
value 12 15 u8 1
value 13 516 f32 -1.5
sleep 4000

# --- Block B: handles 2/3/4 for DRS_ZONE/DRS_ACTIVE/DELTA ---
value 6 1 i16 31
value 7 4 u8 6
value 8 5 f32 29.0
value 9 9 i32 60
value 2 14 u8 1
value 3 15 u8 1
value 4 516 f32 2.5
sleep 4000
