# Page 4 diagnostic probe.
# Step 1: Confirm page 1 renders known values (hardware sanity check).
# Step 2: Switch to page 4 and try values with a generous settle.
# Step 3: Repeatedly resend values in case the first send was lost.
# Run with: script scripts\page4-diag.txt

connect
activate on

# --- Step 1: Page 1 sanity check ---
# Expect: Speed=65, Gear=3, Lap=5/20, Pos=2/20, LapTime=1:05.40, LastLap=1:04.10
pageset 3 1
paramdefs 82 00 00 /0 83 00 00 /0 84 00 00 - 85 00 00 -
keepalive
sleep 500
value 0 1 i16 65
value 1 4 u8 3
value 2 505 u8 5
value 3 501 u8 2
value 4 509 f32 65.4
value 5 510 f32 64.1
sleep 4000

# --- Step 2: Switch to page 4, long settle ---
pageset 3 4
paramdefs 88 00 00 - 88 01 00 - 88 02 00 - 88 03 00 -
keepalive
sleep 2000

# --- Step 3: Send values — expect LastLap ~1:35.50, BestLap ~1:34.20, Ahead 2.35, Behind 1.87 ---
value 0 1 i16 65
value 1 4 u8 3
value 2 510 f32 95.5
value 3 511 f32 94.2
value 4 519 f32 2.35
value 5 520 f32 1.87
sleep 3000

# Resend in case the first batch was lost
value 2 510 f32 95.5
value 3 511 f32 94.2
value 4 519 f32 2.35
value 5 520 f32 1.87
sleep 3000

# --- Step 4: Change values to confirm screen is live ---
value 2 510 f32 102.7
value 3 511 f32 101.3
value 4 519 f32 0.45
value 5 520 f32 5.10
sleep 4000
